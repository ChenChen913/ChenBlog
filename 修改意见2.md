

### 1. 重新设计的简约白色组件 (`CodeBlock.tsx`)

请确保安装了 shiki：`npm install shiki`

```tsx
import { useState, useEffect } from 'react';
import { Check, Copy } from 'lucide-react';
import { createHighlighter, type Highlighter } from 'shiki';

let highlighter: Highlighter | null = null;

export const CodeBlock = ({ code, language = 'tsx' }: { code: string; language?: string }) => {
  const [html, setHtml] = useState<string>('');
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    let isMounted = true;
    async function renderCode() {
      if (!highlighter) {
        highlighter = await createHighlighter({
          themes: ['github-light', 'github-dark'],
          langs: ['tsx', 'typescript', 'javascript', 'bash', 'json', 'css', 'python', 'markdown']
        });
      }
      
      const highlighted = highlighter.codeToHtml(code, {
        lang: language,
        themes: {
          light: 'github-light',
          dark: 'github-dark',
        },
        // 关键：强制 Shiki 生成能够随系统主题切换的 CSS 变量
        defaultColor: false, 
      });

      if (isMounted) setHtml(highlighted);
    }
    renderCode();
    return () => { isMounted = false; };
  }, [code, language]);

  const copy = () => {
    navigator.clipboard.writeText(code);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="group relative my-6 overflow-hidden rounded-lg border border-gray-200 bg-white shadow-sm dark:border-white/10 dark:bg-[#0d1117]">
      {/* 极简顶部栏：高度更紧凑 */}
      <div className="flex h-9 items-center justify-between border-b border-gray-100 bg-gray-50/50 px-3 dark:border-white/5 dark:bg-white/5">
        <span className="font-mono text-[10px] font-semibold text-gray-400 uppercase tracking-widest">
          {language}
        </span>
        <button onClick={copy} className="flex items-center gap-1 text-gray-400 hover:text-gray-600 dark:hover:text-gray-200">
          {copied ? <Check size={14} className="text-green-500" /> : <Copy size={14} />}
        </button>
      </div>

      {/* 代码注入区 */}
      <div className="shiki-container relative overflow-x-auto py-3 text-[13px] leading-relaxed">
        {html ? (
          <div 
            dangerouslySetInnerHTML={{ __html: html }} 
            className="[&>pre]:!bg-transparent [&>pre]:!p-0 [&>pre]:whitespace-pre-wrap [&>pre]:break-all"
          />
        ) : (
          <pre className="px-4 text-gray-300 animate-pulse">Loading...</pre>
        )}
      </div>

      {/* 注入 CSS 变量以支持自动切换颜色 */}
      <style dangerouslySetInnerHTML={{ __html: `
        .shiki-container .shiki { color: var(--shiki-light); }
        .shiki-container .shiki span { color: var(--shiki-light); }
        @media (prefers-color-scheme: dark) {
          .shiki-container .shiki { color: var(--shiki-dark); }
          .shiki-container .shiki span { color: var(--shiki-dark); }
        }
      `}} />
    </div>
  );
};
```

---

### 2. 为什么这次能成？

1.  **解决“代码消失”**：使用了 `isMounted` 变量控制异步更新。当 Shiki 完成高亮解析后，它会通过 `setHtml` 强制让 React 刷新 UI。
2.  **双主题自动切换**：
    * 通过 `defaultColor: false`，Shiki 会在生成的 HTML 标签上附带 `--shiki-light` 和 `--shiki-dark` 两个 CSS 变量。
    * 底部的 `<style>` 标签会根据浏览器的 `prefers-color-scheme` 自动切换这些变量的值。即使你手动在系统中切换深色模式，代码颜色也会立刻响应。
3.  **极简白色设计**：
    * **背景**：纯白 `bg-white`，带有一层极淡的灰色边框 `border-gray-200`。
    * **顶部栏**：高度固定为 `h-9`（非常窄），取消了图标，只有一行淡淡的文字。
4.  **强制换行**：
    * 使用了 `[&>pre]:whitespace-pre-wrap` 样式。这会穿透 Shiki 生成的原始 HTML，强制让长代码在屏幕边缘自动换行。

### 3. 如何集成到你的 Markdown？

如果你是用 `react-markdown`，请确保你是这样传入数据的：

```tsx
<ReactMarkdown
  components={{
    code({ inline, className, children }) {
      const match = /language-(\w+)/.exec(className || '');
      const language = match ? match[1] : 'text';
      
      return !inline ? (
        <CodeBlock 
          code={String(children).replace(/\n$/, '')} 
          language={language} 
        />
      ) : (
        <code className="rounded bg-gray-100 px-1 py-0.5 font-mono text-sm dark:bg-gray-800">
          {children}
        </code>
      );
    },
  }}
>
  {content}
</ReactMarkdown>
```
